cmake_minimum_required(VERSION 3.30)
project(RecastNavigation VERSION 2.0.0 LANGUAGES CXX)

option(RECASTNAVIGATION_MODERN_MODULES "Compile the C++23 named-module API" ON)
option(RECASTNAVIGATION_UPSTREAM_ORACLE "Compile original algorithm targets for differential verification" OFF)
option(RECASTNAVIGATION_TESTS "Compile the original differential verification tests" OFF)
option(RECASTNAVIGATION_DT_POLYREF64 "Use the original oracle's 64-bit serialized refs" OFF)
option(RECASTNAVIGATION_DT_VIRTUAL_QUERYFILTER "Use the original oracle's virtual query filter" OFF)
set(RECASTNAVIGATION_ENABLE_ASSERTS "$<CONFIG:Debug>" CACHE STRING "Oracle assertion condition")

if(RECASTNAVIGATION_MODERN_MODULES)
    add_subdirectory(Modern)
endif()

if(RECASTNAVIGATION_UPSTREAM_ORACLE)
    include(GNUInstallDirs)
    set(SOVERSION 1)
    set(LIB_VERSION 1.6.0)
    string(REPLACE "." "," LIB_VERSION_NUM "${LIB_VERSION}.0")
    configure_file(version.h.in version.h)
    foreach(component IN ITEMS Recast Detour DetourTileCache DetourCrowd DebugUtils)
        add_subdirectory(${component})
        target_compile_features(${component} PRIVATE cxx_std_23)
        if(MSVC)
            target_compile_options(${component} PRIVATE /W3 /WX /wd4351)
            target_compile_definitions(${component} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${component} PRIVATE -Wall -Wextra -Werror)
        endif()
    endforeach()
    if(RECASTNAVIGATION_TESTS)
        enable_testing()
        add_subdirectory(Tests)
    endif()
elseif(RECASTNAVIGATION_TESTS)
    message(FATAL_ERROR "Original differential tests require RECASTNAVIGATION_UPSTREAM_ORACLE")
endif()
